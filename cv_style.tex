% ---------- Competence model ----------
% Reusable key-value object for competence entries.

\ExplSyntaxOn

\tl_new:N \l_cv_competence_title_tl
\tl_new:N \l_cv_competence_skills_tl

\keys_define:nn { cv / competence }
  {
    title  .tl_set:N = \l_cv_competence_title_tl,
    skills .tl_set:N = \l_cv_competence_skills_tl,
  }

\cs_new_protected:Npn \cv_competence_clear:
  {
    \tl_clear:N \l_cv_competence_title_tl
    \tl_clear:N \l_cv_competence_skills_tl
  }

\cs_new_protected:Npn \cv_competence_print:
  {
    \par\noindent
    {\CVCompetenceTitleFont
     \tl_use:N \l_cv_competence_title_tl
     \textnormal{:}}
    ~
    {\CVCompetenceSkillsFont
    \enspace
     \tl_use:N \l_cv_competence_skills_tl}
    \par
  }

\NewDocumentCommand{\Competence}{m}
  {
    \cv_competence_clear:
    \keys_set:nn { cv / competence } { #1 }
    \cv_competence_print:
  }

\ExplSyntaxOff

% ---------- Work experience model ----------
% Work entries pass raw \item lines through the `items` key.
% The model owns the cvitemlist wrapper so bullet spacing can be changed once.

\ExplSyntaxOn

\tl_new:N \l_cv_work_title_tl
\tl_new:N \l_cv_work_title_regular_tl
\tl_new:N \l_cv_work_organization_tl
\tl_new:N \l_cv_work_dates_tl
\tl_new:N \l_cv_work_location_tl
\tl_new:N \l_cv_work_description_tl
\tl_new:N \l_cv_work_items_tl

\keys_define:nn { cv / work }
  {
    title        .tl_set:N = \l_cv_work_title_tl,
    title_regular .tl_set:N = \l_cv_work_title_regular_tl,
    organization .tl_set:N = \l_cv_work_organization_tl,
    dates        .tl_set:N = \l_cv_work_dates_tl,
    location     .tl_set:N = \l_cv_work_location_tl,
    description  .tl_set:N = \l_cv_work_description_tl,
    items        .tl_set:N = \l_cv_work_items_tl,
  }

\cs_new_protected:Npn \cv_work_clear:
  {
    \tl_clear:N \l_cv_work_title_tl
    \tl_clear:N \l_cv_work_title_regular_tl
    \tl_clear:N \l_cv_work_organization_tl
    \tl_clear:N \l_cv_work_dates_tl
    \tl_clear:N \l_cv_work_location_tl
    \tl_clear:N \l_cv_work_description_tl
    \tl_clear:N \l_cv_work_items_tl
  }

\cs_new_protected:Npn \cv_work_print_title:
  {
    \CVTableHeadingFont
    {\bfseries \tl_use:N \l_cv_work_title_tl}
    \tl_if_empty:NF \l_cv_work_title_regular_tl
      {, ~\tl_use:N \l_cv_work_title_regular_tl}
  }

\cs_new_protected:Npn \cv_work_print:
  {
    \Needspace{8\baselineskip}
    \begingroup
    \setlength{\parskip}{0pt}
    \setlength{\tabcolsep}{2pt}
    \renewcommand{\arraystretch}{1.0}

    \par\noindent
    \begin{tabularx}{\textwidth}{
      B
      >{\raggedleft\arraybackslash}b{\CVTableMetaColumnWidth}
    }
      {\cv_work_print_title:} &
      {\CVTableMetaFont\itshape\color{cvTableMetaTextColor}\tl_use:N \l_cv_work_dates_tl} \\

      {\CVTableMetaFont\itshape\color{cvTableMetaTextColor}\tl_use:N \l_cv_work_organization_tl} &
      {\CVTableMetaFont\itshape\color{cvTableMetaTextColor}\tl_use:N \l_cv_work_location_tl} \\
    \end{tabularx}
    \par

    \tl_if_empty:NF \l_cv_work_description_tl
      {
        \vspace{0.5em}
        \CVTableBodyFont\tl_use:N \l_cv_work_description_tl
      }

    \vspace{0.5em}

    {\CVTableBodyFont
      \begin{cvitemlist}
        \tl_use:N \l_cv_work_items_tl
      \end{cvitemlist}
    }

    \endgroup
  }

\NewDocumentCommand{\WorkEntry}{m}
  {
    \cv_work_clear:
    \keys_set:nn { cv / work } { #1 }
    \cv_work_print:
  }
