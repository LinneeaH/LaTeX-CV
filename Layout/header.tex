\newcommand{\CVHeader}{%
  \sbox{\CVHeaderContentBox}{\CVHeaderContent}%
  \setlength{\CVHeaderTextLeft}{\dimexpr 1in+\hoffset+\oddsidemargin\relax}%
  \setlength{\CVHeaderTextTop}{\dimexpr 1in+\voffset+\topmargin+\headheight+\headsep\relax}%
  \setlength{\CVHeaderContentBottom}{\dimexpr \CVHeaderContentTop+\ht\CVHeaderContentBox+\dp\CVHeaderContentBox+\CVHeaderContentBottomPadding\relax}%
  \setlength{\CVHeaderReservedBottom}{\CVHeaderHeight}%
  \ifdim\CVHeaderContentBottom>\CVHeaderReservedBottom
    \setlength{\CVHeaderReservedBottom}{\CVHeaderContentBottom}%
  \fi
  \begin{tikzpicture}[remember picture,overlay]
    \path[
      left color=cvHeaderGradientStart,
      right color=cvHeaderGradientEnd
    ] (current page.north west) rectangle ([yshift=-\CVHeaderReservedBottom]current page.north east);
    \draw[black,line width=0.45pt]
      ([yshift=-\CVHeaderReservedBottom]current page.north west) --
      ([yshift=-\CVHeaderReservedBottom]current page.north east);
    \node[anchor=north west,inner sep=0pt,align=left]
      at ([xshift=\CVHeaderTextLeft,yshift=-\CVHeaderContentTop]current page.north west)
      {\usebox{\CVHeaderContentBox}};
  \end{tikzpicture}%
  \ifdim\CVHeaderReservedBottom>\CVHeaderTextTop
    \vspace*{\dimexpr \CVHeaderReservedBottom-\CVHeaderTextTop\relax}%
  \fi
}
